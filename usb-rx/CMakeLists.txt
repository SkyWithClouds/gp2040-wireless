cmake_minimum_required(VERSION 3.20.0)
set(BRIDGE_STAGE "xinput" CACHE STRING "Bridge integration stage")
if(BRIDGE_STAGE MATCHES "^(esb|full)$")
  set(EXTRA_CONF_FILE esb.conf)
elseif(NOT BRIDGE_STAGE STREQUAL "xinput")
  message(FATAL_ERROR "The receiver supports xinput, esb and full")
endif()
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(bridge_xinput)
# Pinned legacy USB stack supplies OS 1.0 registration and descriptor fixup.
target_include_directories(app PRIVATE ${ZEPHYR_BASE}/subsys/usb/device)
target_sources(app PRIVATE src/main.c src/reports.c)
if(BRIDGE_STAGE MATCHES "^(esb|full)$")
  target_compile_definitions(app PRIVATE BRIDGE_ESB=1)
  target_include_directories(app PRIVATE ../common)
  target_sources(app PRIVATE ../common/air.c src/receiver.c)
endif()
